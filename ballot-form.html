Make the Google Form

This script builds the whole thing. Four steps, about two minutes.

1. Go to forms.new

2. Extensions → Apps Script

3. Paste this over everything that is there. Save. Run.

3b. Check the dropdown next to Debug says buildTheBallot, not soft. Apps Script picks the first function and running the wrong one completes silently with no output, which looks like it worked.

4. Approve it. “Google hasn't verified this app” shows up because you wrote it — Advanced → Go to (unsafe). Your link prints at the bottom.

/**
 * Builds the October 18 ballot.
 *
 *   1. Go to forms.new
 *   2. Extensions > Apps Script
 *   3. Paste ALL of this over whatever is there, Save, then Run.
 *   4. Approve it. "Google hasn't verified this app" appears because YOU
 *      wrote it -- Advanced > Go to (unsafe). It is your own script.
 *   5. Your link prints in the Execution log at the bottom.
 *
 * Safe to run twice: it clears the form first.
 */

// NOTE: buildTheBallot MUST be the first function in this file. Apps Script
// preselects the first one in the Run dropdown, and running the helper by
// mistake completes silently with no output, which looks like success.

function buildTheBallot() {
  var form = null;
  try { form = FormApp.getActiveForm(); } catch (e) { form = null; }
  if (!form) { form = FormApp.create("Wall Street Warriors — October ballot"); }

  form.getItems().forEach(function (item) { form.deleteItem(item); });
  form.setTitle("Wall Street Warriors — October ballot");
  form.setDescription("Nine questions. Five need 4 of 5 to carry, which is why meetings with partial attendance keep failing to settle them.\n\nCloses Wednesday Oct 21. Reasoning for every question: screenstakes.github.io/wall-street-warriors/ballot.html");

  // setPublishingSummary(false) is the one that matters. Leave it on and
  // every responder can read everyone else's answers in full.
  soft(function () { form.setPublishingSummary(false); });
  soft(function () { form.setCollectEmail(false); });
  soft(function () { form.setProgressBar(true); });
  soft(function () { form.setAllowResponseEdits(true); });
  soft(function () { form.setShowLinkToRespondAgain(false); });
  soft(function () { form.setConfirmationMessage("Got it. Result posted Wednesday, win or lose."); });
  soft(function () { form.setPublished(true); });

  form.addMultipleChoiceItem()
    .setTitle('Who are you?')
    .setChoiceValues(["Michael", "Caleb", "Dylan", "Joe", "Federico"])
    .setRequired(true);

  form.addMultipleChoiceItem()
    .setTitle("Ratify the Pile C trigger, as written?")
    .setHelpText("Written Oct 5, put on our site, never voted on. Needs 4 of 5.")
    .setChoiceValues(["Yes, as written", "Yes, but change something", "No"])
    .setRequired(true);

  form.addMultipleChoiceItem()
    .setTitle("Cap any one theme at 20% of the portfolio?")
    .setHelpText("Our AI chain is 19.2% once you count what VOO already holds. Needs 4 of 5.")
    .setChoiceValues(["Yes, 20%", "Yes, but a different number", "No cap — just measure it"])
    .setRequired(true);

  form.addMultipleChoiceItem()
    .setTitle("Adopt this sell rule?")
    .setHelpText("We sell when the reason no longer fits the company, a weight cap breaks, or we need the money. Not because the price fell. Needs 4 of 5.")
    .setChoiceValues(["Yes, as written", "Yes, with changes", "No"])
    .setRequired(true);

  form.addMultipleChoiceItem()
    .setTitle("Pile C: leave it, or cut it to 10%?")
    .setHelpText("It is 13.72% now. Cutting frees about $11,000. Needs 4 of 5.")
    .setChoiceValues(["Leave it", "Cut to 10%, into VEA", "Cut to 10%, hold as cash"])
    .setRequired(true);

  form.addMultipleChoiceItem()
    .setTitle("Ratify the BIL trim?")
    .setHelpText("We voted 15%. It went in at 13.72% because the bonds cost more. Never ratified. Needs 4 of 5.")
    .setChoiceValues(["Yes", "No — buy BIL back to 15%"])
    .setRequired(true);

  form.addMultipleChoiceItem()
    .setTitle("Medtronic: hold or sell?")
    .setHelpText("Its diabetes business was split off Oct 9. It is up 9.19%.")
    .setChoiceValues(["Hold", "Sell", "Decide after Nov 6"])
    .setRequired(true);

  form.addMultipleChoiceItem()
    .setTitle("Who submits Friday if Federico has not confirmed by Tuesday?")
    .setHelpText("Due Fri Oct 23, 4:00pm. Everything is already written.")
    .setChoiceValues(["Federico", "Michael", "Caleb"])
    .setRequired(true);

  form.addPageBreakItem().setTitle('Your buy prices').setHelpText('The price you would add at. Rough is fine. Leave blank if you have no view.');
  form.addTextItem().setTitle("GEV");
  form.addTextItem().setTitle("JNJ");
  form.addTextItem().setTitle("NVDA");
  form.addTextItem().setTitle("BKNG");
  form.addTextItem().setTitle("GOOG");
  form.addTextItem().setTitle("IBKR");
  form.addTextItem().setTitle("MDT");
  form.addTextItem().setTitle("ETN");
  form.addTextItem().setTitle("AMZN");

  form.addCheckboxItem()
    .setTitle('Which of these do you actually remember voting on?')
    .setHelpText('Tick only what you remember. Not remembering is a useful answer.')
    .setChoiceValues(["Three piles, 60 / 25 / 15", "Treasuries chosen by maturity date", "Ten stocks, two per sector", "No REIT, no corporates, no Taiwan", "'Certainty' means 97%"]);

  form.addParagraphTextItem()
    .setTitle('Anything you disagree with, or that I got wrong?')
    .setHelpText('This box is the point. A ballot where everyone picks option one proves nothing.');

  var live = form.getPublishedUrl();
  var short = live;
  soft(function () { short = form.shortenFormUrl(live); });
  Logger.log('SHARE THIS LINK:  ' + short);
  Logger.log('Edit it here:     ' + form.getEditUrl());
  Logger.log('Questions added:  ' + form.getItems().length);
}

function soft(fn) { try { fn(); } catch (e) {} }

Three things it does that matter

Answers stay private

setPublishingSummary(false) is the only setting that does this. Leave it on and every person who submits can read everyone else's answers in full. That is the default.

Safe to run twice

It clears the form before building, so a second run does not double every question.

People can change their mind

Response editing is on. You want the final view, not the first one.

What is in it

  • Seven multiple-choice questions — the trigger, the 20% cap, the sell rule, Pile C's size, the BIL trim, Medtronic, who submits Friday. Five of them say “Needs 4 of 5” right in the question.
  • Nine price boxes, one per name. Blank is allowed.
  • One checkbox question: which “Settled” items do you actually remember voting on.
  • One free-text box at the end: “Anything you disagree with, or that I got wrong?” That box is the point.
